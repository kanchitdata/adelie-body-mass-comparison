การทดสอบการแจกแจงปกติ

ใช้การทดสอบของ Shapiro-Wilk กำหนดระดับนัยสำคัญ \(\alpha = 0.05\) \[ \begin{aligned} H_0 &: \text{น้ำหนักตัวของเพนกวิน Adelie ทุกกลุ่มย่อยมีการแจกแจงปกติ} \\ H_1 &: \text{มีอย่างน้อย 1 กลุ่มที่การแจกแจงไม่เป็นแบบปกติ} \end{aligned} \]

Code 
adelie %>% 
  summarise(
    W = shapiro.test(body_mass_g)$statistic,
    p_value = shapiro.test(body_mass_g)$p.value,
            .by= c(island, sex)) %>% 
  gt()

จากตารางที่ 1 พบว่าทุกกลุ่มมีค่า p-value มากกว่า 0.05 จึงไม่ปฏิเสธ \(H_0\) สรุปได้ว่าน้ำหนักตัวของเพนกวิน Adelie ในทุกกลุ่มย่อยมีการแจกแจงปกติ


การทดสอบความเท่ากันของความแปรปรวน

การทดสอบความเท่ากันของความแปรปรวนระหว่างตัวผู้และตัวเมียในเกาะเดียวกันในแต่ละเกาะ กำหนด\(\alpha = 0.05\)

\[ \begin{aligned} H_0 &: \text{ความแปรปรวน body\_mass\_g ของเพนกวินสายพันธ์ Adelie ตัวผู้และตัวเมีย ในเกาะเดียวกันเท่ากันทุกเกาะ} \\ H_1 &: \text{ความแปรปรวน body\_mass\_g ของเพนกวินสายพันธ์ Adelie ตัวผู้และตัวเมีย ในเกาะเดียวกันมีอย่างน้อย 1 กลุ่มที่แตกต่างกัน} \end{aligned} \]

Code 
ftest.mf <- function(isl) {
return(
  adelie %>% 
  filter(island == isl) %>% 
  { var.test(body_mass_g ~ sex,data=.)} %>% 
  tidy()
)
  
}

c("Torgersen", "Dream", "Biscoe") %>%
  set_names() %>%
  map_dfr(ftest.mf, .id = "island") %>% 
select(-c(method, conf.low, conf.high)) %>% 
  mutate(island = c('Torgersen', 'Dream', 'Biscoe')) %>% 
  select(island, everything()) %>% 
  gt()

จากการทดสอบ ftest-test พบว่า p-value ของเกาะ Torgersen และ Biscoe มากกว่า 0.05 ยกเว้นเกาะ Dream ดังนั้นจึงตัดสินใจปฏิเสธสมติฐานหลักและยอมรับสมติฐานแย้งความแปรปรวน body_mass_g ของเพนกวินสายพันธ์ Adelie ตัวผู้และตัวเมีย ในเกาะเดียวกัน มีอย่างน้อย 1 กลุ่มที่แตกต่างกัน นั้นก็คือเกาะ Dream

pair.ftest.adelie <- function(var1, var2) {

    adelie %>%

  tidy()

การทดสอบค่าเฉลี่ยของน้ำหนักตัว

  • ก่อนจะไปทำการทดสอบว่าค่าเฉลี่ยเพนกวินในแต่ละเกาะต่างกันไหม จะทำการตรวจสอบว่าค่าเฉลี่ยน้ำหนักของตัวผู้และตัวเมียแตกต่างกันหรือไม่ เพื่อตัดสินใจที่จะเปรียบเทียบรวมทั้งเพศเมียและเพศผู้ด้วยเลย เมื่อค่าเฉลี่ยของตัวผู้และตัวเมียในแต่ละเกาะไม่แตกต่างกัน แต่ถ้าค่าเฉลี่ยของตัวผู้และตัวเมียแตกต่างกัน จะทำการเปรียบเทียบแบบแต่ละเกาะแต่ แยกเพศ

จากการทดสอบการแจกแจงปกติและความเท่ากันของความแปรปรวนทำให้เรารู้ว่า

  • การแจงแจงของทุกกลุ่มมีการแจกแจงปกติ

  • ความแปรปรวนในเพศผู้แต่คนละเกาะเท่ากันในทั้ง 3 เกาะ แต่สำหรับเพศเมียมี 2 คู่ที่ความแปรปรวนเท่ากันคือ เกาะ Torgersen Dream และ Torgersen Biscoe ส่วนคู่ที่ไม่เท่ากันคือ Dream biscoe

การทดสอบความเท่ากันของค่าเฉลี่ยระหว่างเพนกวินเพศผู้และเพนกวินเพศเมียในเกาะเดียวกัน

  • การแจกแจงของทุกกลุ่มที่จะนำมาเปรียบเทียบเป็นการแจกแจงปกติ

  • ความเท่ากันของความแปรปรวนระหว่างเพนกวินเพศผู้และเพศเมียในเกาะเดียวกันมีความแปรปรวนเท่ากันยกเว้นเกาะ Dream ที่มีความแปรปรวนไม่เท่ากันระหว่างเพนกวินเพศผู้และเพศเมีย

สำหรับวิธีการเปรียบเทียบค่าเฉลี่ย

  • t-test ความแปรปรวนเท่ากันสำหรับเกาะ Torgersen และ Biscoe

  • t-test ความแปรวปรวนไม่เท่ากันสำหรับเกาะ Dream

Code 
t.test.mf <- function(isl, var.e=F) {
male.m <- adelie %>% 
  filter(island == isl, sex=='male') %>% 
  pull(body_mass_g)

female.m <- adelie %>% 
  filter(island == isl, sex=="female") %>% 
  pull(body_mass_g)

result <- t.test(male.m, female.m, var.equal=var.e, alternative="two.sided") %>% 
  tidy()

return(result)
}
