\[ \begin{aligned} H_0 &: \text{น้ำหนักตัวของเพนกวิน Adelie ทุกกลุ่มย่อยมีการแจกแจงปกติ} \\ H_1 &: \text{น้ำหนักตัวของเพนกวิน Adelie มีอย่างน้อย 1 กลุ่มที่การแจกแจงไม่เป็นแบบปกติ} \end{aligned} \]

การทดสอบความเท่ากันของความแปรปรวน

การทดสอบความเท่ากันของความแปรปรวนของ body_mass_g ระหว่างตัวผู้และตัวเมียในเกาะเดียวกันในแต่ละเกาะด้วย F-test

กำหนดระดับนัยสำคัญ 0.05

\[ \begin{aligned} % เกาะ Torgersen H_{01} &: \sigma^2_{\text{ผู้, Torgersen}} = \sigma^2_{\text{เมีย, Torgersen}} \\ H_{11} &: \sigma^2_{\text{ผู้, Torgersen}} \neq \sigma^2_{\text{เมีย, Torgersen}} \\[1ex] % เกาะ Dream H_{02} &: \sigma^2_{\text{ผู้, Dream}} = \sigma^2_{\text{เมีย, Dream}} \\ H_{12} &: \sigma^2_{\text{ผู้, Dream}} \neq \sigma^2_{\text{เมีย, Dream}} \\[1ex] % เกาะ Biscoe H_{03} &: \sigma^2_{\text{ผู้, Biscoe}} = \sigma^2_{\text{เมีย, Biscoe}} \\ H_{13} &: \sigma^2_{\text{ผู้, Biscoe}} \neq \sigma^2_{\text{เมีย, Biscoe}} \end{aligned} \]

Code 
ftest.mf <- function(isl) {
return(
  adelie %>% 
  filter(island == isl) %>% 
  { var.test(body_mass_g ~ sex,data=.)} %>% 
  tidy()
)
  
}

c(Torgersen = "Torgersen", Dream = "Dream", Biscoe = "Biscoe") %>% 
  map_dfr(ftest.mf, .id = "island") %>% 
select(-c(method, conf.low, conf.high, alternative)) %>% 
  select(island, everything()) %>% 
  gt()

จากการทดสอบความแปรปรวนเท่ากันด้วย F-test ที่ระดับความสำคัญทางสถิติ \(\alpha = 0.05\) พบว่า เกาะ Torgersen และเกาะ Biscoe มี \(p\)-value มากกว่า \(0.05\) ในขณะที่เกาะ Dream มี \(p\)-value น้อยกว่า \(0.05\)

จึงทำการตัดสินใจ:

  • ยอมรับ \(H_{01}\) และ \(H_{03}\) สำหรับเกาะ Torgersen และ Biscoe ตามลำดับ สรุปได้ว่าความแปรปรวนของน้ำหนักตัว (\(body\_mass\_g\)) ระหว่างเพศผู้และเพศเมียในสองเกาะนี้ไม่แตกต่างกัน

  • ปฏิเสธ \(H_{02}\) สำหรับเกาะ Dream สรุปได้ว่าความแปรปรวนของน้ำหนักตัว (\(body\_mass\_g\)) ระหว่างเพศผู้และเพศเมียในเกาะ Dream มีความแตกต่างกัน อย่างมีนัยสำคัญทางสถิติ

ข้อสรุปภาพรวม: ความแปรปรวนของน้ำหนักตัวเพนกวินสายพันธุ์ Adelie ระหว่างเพศผู้และเพศเมียมีความแตกต่างกันอย่างน้อย 1 เกาะ ซึ่งก็คือ เกาะ Dream


การทดสอบความเท่ากันของความแปรปรวน body_mass_g เพศผู้ในแต่ละเกาะด้วยวิธีการ Levene-test

กำหนดระดับนัยสำคัญ 0.05

\[ \begin{aligned} H_0 &: \sigma^2_{\text{ตัวผู้, Torgersen}} = \sigma^2_{\text{ตัวผู้, Dream}} = \sigma^2_{\text{ตัวผู้, Bisoe}} \\ H_1 &: \text{มีอย่างน้อย 1 กลุ่มที่แตกต่างกัน} \end{aligned} \]

กำหนดระดับสำคัญ 0.05

\[ \begin{aligned} H_0 &: \sigma^2_{\text{ตัวเมีย, Torgersen}} = \sigma^2_{\text{ตัวเมีย, Dream}} = \sigma^2_{\text{ตัวเมีย, Bisoe}} \\ H_1 &: \text{มีอย่างน้อย 1 กลุ่มที่แตกต่างกัน} \end{aligned} \]

หลังจากที่เรารู้แล้วว่ามีความแปรรวนอย่างน้อย 1 ตัวไม่เท่ากับความแปรปรวนของกลุ่มอื่นดังนั้นจะใช้การทดสอบความเท่ากันของความแปรปรวน F-test ในการตรวจว่าความแปรปรวนในกลุ่มไหนที่ไม่เหมือนกัน

pair.ftest.adelie <- function(var1, var2) {
  return(
    adelie %>% 
  filter(sex == "female", island %in% c(var1, var2)) %>% 
  {var.test(body_mass_g ~ island,data=.)} %>% 
  tidy()
  )
}

สรุปผล จากการทดสอบความแปรปรวนเท่ากันด้วย F-test ที่ระดับความสำคัญทางสถิติ \(\alpha = 0.05\) พบว่า เกาะ Torgersen vs Dream และ Torgersen vs Biscoe มีค่า p-value มากกว่า \(0.05\) ในขณะที่เกาะ Dream vs Biscoe มีค่า p-value น้อยกว่า \(0.05\)

จึงทำการตัดสินใจ:

  • ยอมรับ \(H_{01}\) และ \(H_{02}\) สำหรับเกาะ Torgersen vs Dream และ Torgersen vs Biscoe ตามลำดับ สรุปได้ว่าความแปรปรวนของน้ำหนักตัว (\(body\_mass\_g\)) ไม่แตกต่างกัน

  • ปฏิเสธ \(H_{03}\) สำหรับเกาะ Dream vs Biscoe สรุปได้ว่าความแปรปรวนของน้ำหนักตัว (\(body\_mass\_g\)) มีความแตกต่างกัน อย่างมีนัยสำคัญทางสถิติ

ข้อสรุปภาพรวม: ความเท่ากันของความแปรปรวนน้ำหนักเพนกวินเพศเมีย มีเกาะที่แตกต่างกันคือเกาะ Dream และ Biscoe


การทดสอบค่าเฉลี่ยของน้ำหนักตัว

  • ก่อนจะไปทำการทดสอบว่าค่าเฉลี่ยเพนกวินในแต่ละเกาะต่างกันไหม จะทำการตรวจสอบว่าค่าเฉลี่ยน้ำหนักของตัวผู้และตัวเมียแตกต่างกันหรือไม่ เพื่อตัดสินใจที่จะเปรียบเทียบรวมทั้งเพศเมียและเพศผู้ด้วยเลย เมื่อค่าเฉลี่ยของตัวผู้และตัวเมียในแต่ละเกาะไม่แตกต่างกัน แต่ถ้าค่าเฉลี่ยของตัวผู้และตัวเมียแตกต่างกัน จะทำการเปรียบเทียบแบบแต่ละเกาะแต่ แยกเพศ

จากการทดสอบการแจกแจงปกติและความเท่ากันของความแปรปรวนทำให้เรารู้ว่า

  • การแจงแจงของทุกกลุ่มมีการแจกแจงปกติ (Table 2)

  • ความแปรปรวนในเพศผู้แต่คนละเกาะเท่ากันในทั้ง 3 เกาะ (Table 4) แต่สำหรับเพศเมียมี 2 คู่ที่ความแปรปรวนเท่ากันคือ เกาะ Torgersen Dream และ Torgersen Biscoe ส่วนคู่ที่ไม่เท่ากันคือ Dream biscoe (Table 6)


การทดสอบความเท่ากันของค่าเฉลี่ยระหว่างเพนกวินเพศผู้และเพนกวินเพศเมียในเกาะเดียวกัน

  • การแจกแจงของทุกกลุ่มที่จะนำมาเปรียบเทียบเป็นการแจกแจงปกติ

  • ความเท่ากันของความแปรปรวนระหว่างเพนกวินเพศผู้และเพศเมียในเกาะเดียวกันมีความแปรปรวนเท่ากันยกเว้นเกาะ Dream ที่มีความแปรปรวนไม่เท่ากันระหว่างเพนกวินเพศผู้และเพศเมีย

สำหรับวิธีการเปรียบเทียบค่าเฉลี่ย

  • t-test ความแปรปรวนเท่ากันสำหรับเกาะ Torgersen และ Biscoe

  • t-test ความแปรวปรวนไม่เท่ากันสำหรับเกาะ Dream


การทดสอบความเท่ากันของค่าเฉลี่ย body_mass_g ระหว่างเพศผู้และเพศเมียในเกาะเดียวกัน

\[ \begin{aligned} % เกาะ Torgersen (var.equal = TRUE) H_{01} &: \mu_{\text{ผู้, Torgersen}} = \mu_{\text{เมีย, Torgersen}} \\ H_{11} &: \mu_{\text{ผู้, Torgersen}} \neq \mu_{\text{เมีย, Torgersen}} \\[1ex] % เกาะ Biscoe (var.equal = TRUE) H_{02} &: \mu_{\text{ผู้, Biscoe}} = \mu_{\text{เมีย, Biscoe}} \\ H_{12} &: \mu_{\text{ผู้, Biscoe}} \neq \mu_{\text{เมีย, Biscoe}} \\[1ex] % เกาะ Dream (var.equal = FALSE) H_{03} &: \mu_{\text{ผู้, Dream}} = \mu_{\text{เมีย, Dream}} \\ H_{13} &: \mu_{\text{ผู้, Dream}} \neq \mu_{\text{เมีย, Dream}} \end{aligned} \]

Code 
t.test.mf <- function(isl, var.e=F) {
male.m <- adelie %>% 
  filter(island == isl, sex=='male') %>% 
  pull(body_mass_g)

female.m <- adelie %>% 
  filter(island == isl, sex=="female") %>% 
  pull(body_mass_g)

result <- t.test(male.m, female.m, var.equal=var.e, alternative="two.sided") %>% 
  tidy()

return(result)
}



map2_dfr(
  .x = c("Torgersen", "Biscoe", "Dream"),
  .y = c(T, T, F),
  .f = t.test.mf
) %>% 
  select(-c(estimate, estimate1, estimate2, conf.low, conf.high)) %>% 
  mutate(
    island = c("Torgersen", "Biscoe", "Dream"),
    p.value = pvalue(p.value)
  ) %>% 
  select(island, everything()) %>% 
  gt()
